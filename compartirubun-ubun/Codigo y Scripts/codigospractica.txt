PARTE 4. SCRIPT BASH DEL SERVIDOR (configurar_servidor.sh)
------------------------------------------------------------------
#!/usr/bin/env bash
# Configura un servidor Samba y comparte una carpeta.
# Uso: bash configurar_servidor.sh [usuario] [nombre_recurso]
set -euo pipefail          # Detiene el script ante cualquier error

# --- Parámetros ---
USUARIO="${1:-$USER}"      # Usuario con acceso SMB
RECURSO="${2:-isramba}"    # Nombre del recurso en la red
ESCRITORIO="$(xdg-user-dir DESKTOP 2>/dev/null || echo "$HOME/Desktop")"
CARPETA="$ESCRITORIO/IsrAMBA"
CONF="/etc/samba/smb.conf"

# --- Funciones ---
instalar_samba() {
    echo "[1/5] Instalando Samba..."
    sudo apt update -qq
    sudo apt install -y samba
}

crear_carpeta() {
    echo "[2/5] Creando la carpeta $CARPETA..."
    mkdir -p "$CARPETA"
}

configurar_recurso() {
    echo "[3/5] Configurando el recurso [$RECURSO]..."
    # Evita duplicar el bloque si el script se ejecuta dos veces
    if grep -q "^\[$RECURSO\]" "$CONF"; then
        echo "      El recurso ya existe."
        return
    fi
    sudo cp "$CONF" "$CONF.bak"          # Copia de seguridad
    sudo tee -a "$CONF" > /dev/null <<BLOQUE

[$RECURSO]
   comment = Carpeta compartida entre Ubuntu
   path = $CARPETA
   browseable = yes
   read only = no
   guest ok = no
   valid users = $USUARIO
BLOQUE
    testparm -s > /dev/null              # Valida la sintaxis
}

crear_usuario_samba() {
    echo "[4/5] Contraseña SMB de $USUARIO..."
    sudo smbpasswd -a "$USUARIO"
}

reiniciar_servicio() {
    echo "[5/5] Reiniciando servicio y firewall..."
    sudo systemctl restart smbd
    sudo ufw allow samba > /dev/null 2>&1 || true
}

# --- Ejecución ---
instalar_samba
crear_carpeta
configurar_recurso
crear_usuario_samba
reiniciar_servicio
echo "IP del servidor: $(hostname -I | awk '{print $1}')"

PARTE 5. SCRIPT BASH DEL CLIENTE (conectar_cliente.sh)

#!/usr/bin/env bash
# Se conecta a una carpeta compartida por Samba.
# Uso: bash conectar_cliente.sh <ip_servidor> <usuario> [recurso]
set -euo pipefail

# --- Validación de argumentos ---
if [[ $# -lt 2 ]]; then
    echo "Uso: $0 <ip_servidor> <usuario> [recurso]" >&2
    exit 1
fi

IP="$1"
USUARIO="$2"
RECURSO="${3:-isramba}"
PUNTO_MONTAJE="/mnt/$RECURSO"

# --- Funciones ---
verificar_red() {
    echo "[1/4] Probando conectividad con $IP..."
    if ! ping -c 4 "$IP"; then
        echo "Error: no hay conexión con $IP." >&2
        exit 2
    fi
}

instalar_herramientas() {
    echo "[2/4] Instalando herramientas SMB..."
    sudo apt install -y smbclient cifs-utils
}

listar_recursos() {
    echo "[3/4] Recursos compartidos en $IP:"
    smbclient -L "//$IP" -U "$USUARIO"
}

montar_recurso() {
    echo "[4/4] Montando en $PUNTO_MONTAJE..."
    sudo mkdir -p "$PUNTO_MONTAJE"
    if mountpoint -q "$PUNTO_MONTAJE"; then
        echo "      Ya estaba montado."
    else
        sudo mount -t cifs "//$IP/$RECURSO" "$PUNTO_MONTAJE" \
            -o "username=$USUARIO,uid=$(id -u),gid=$(id -g)"
    fi
    ls -la "$PUNTO_MONTAJE"
}

# --- Ejecución ---
verificar_red
instalar_herramientas
listar_recursos
montar_recurso
